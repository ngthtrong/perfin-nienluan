\newcommand{\reportfigure}[4][0.99\linewidth]{%
  \clearpage
  \begin{figure}[p]
    \centering
    \includegraphics[width=#1,height=0.86\textheight,keepaspectratio]{#2}
    \caption{#3}
    \label{#4}
  \end{figure}
  \clearpage}

  \clearpage}

% Viền trang bìa: hai nét bao quanh toàn trang — nét ngoài đậm, nét trong nhạt.
% Dùng overlay TikZ theo kích thước giấy để độc lập với lề nội dung.
\newcommand{\coverborder}{%
  \begin{tikzpicture}[remember picture,overlay]
    % Nét ngoài đậm.
    \draw[line width=1.5pt]
      ($(current page.south west)+(1.2cm,1.2cm)$)
      rectangle
      ($(current page.north east)+(-1.2cm,-1.2cm)$);
    % Nét trong nhạt.
    \draw[line width=0.5pt]
      ($(current page.south west)+(1.5cm,1.5cm)$)
      rectangle
      ($(current page.north east)+(-1.5cm,-1.5cm)$);
  \end{tikzpicture}}
